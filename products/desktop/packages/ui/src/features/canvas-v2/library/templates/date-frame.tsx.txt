// A frame with its own date range. Every data fragment inside it reads this
// range in place of the board range, so one board can show the same numbers
// over two periods. Move the frame and the contents follow.
import { useContainer, useOwnedDateRange } from "@posthog/canvas-sdk";

const PADDING = 14;
const HEADER = 38;

const PRESETS = [
  { value: "-24h", short: "24h", label: "Last 24 hours" },
  { value: "-7d", short: "7d", label: "Last 7 days" },
  { value: "-30d", short: "30d", label: "Last 30 days" },
  { value: "-90d", short: "90d", label: "Last 90 days" },
  { value: "-12m", short: "12m", label: "Last 12 months" },
];

export default function DateFrame({ fragmentId }) {
  const { range, setRange } = useOwnedDateRange(fragmentId);
  const { children } = useContainer(fragmentId, {
    padding: PADDING,
    header: HEADER,
    follow: true,
  });
  const known = PRESETS.some((preset) => preset.value === range.date_from);
  const current = known ? range.date_from : "-7d";

  return (
    <div className="flex h-full w-full flex-col rounded-xl border border-border border-dashed bg-muted/20">
      <div
        className="flex shrink-0 items-center gap-1.5 px-2.5"
        style={{ height: HEADER }}
      >
        <div className="flex items-center gap-0.5 rounded-full border border-border bg-card p-0.5">
          {PRESETS.map((preset) => {
            const selected = preset.value === current;
            return (
              <button
                key={preset.value}
                type="button"
                aria-label={preset.label}
                aria-pressed={selected}
                className={
                  selected
                    ? "rounded-full bg-primary px-2 py-0.5 font-medium text-[11px] text-primary-foreground"
                    : "rounded-full px-2 py-0.5 text-[11px] text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
                }
                onClick={() =>
                  setRange({ date_from: preset.value, date_to: null })
                }
              >
                {preset.short}
              </button>
            );
          })}
        </div>
        <span className="ml-auto shrink-0 text-[11px] text-muted-foreground">
          {children.length === 1 ? "1 fragment" : children.length + " fragments"}
        </span>
      </div>

      {children.length === 0 ? (
        <div className="flex min-h-0 flex-1 items-center justify-center px-4 text-center text-[12px] text-muted-foreground">
          Drag data fragments onto this frame and they follow this date range.
        </div>
      ) : null}
    </div>
  );
}

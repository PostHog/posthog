// Markdown notes. The text is a shared field under STORAGE_KEY, so two people
// can write in it at the same time and no character is lost. For a second
// notes fragment on the same board, change STORAGE_KEY to a different value.
import { SharedTextArea, useSharedText } from "@posthog/canvas-sdk";
import { Button } from "@posthog/quill";
import { useState } from "react";
import Markdown from "react-markdown";

const STORAGE_KEY = "notes:text";

export default function Notes({ storageKey = STORAGE_KEY }) {
  const { text } = useSharedText(storageKey);
  const [editing, setEditing] = useState(false);

  return (
    <div className="flex h-full flex-col gap-2 overflow-auto p-4">
      <div className="flex justify-end">
        <Button
          variant="outline"
          size="sm"
          onClick={() => setEditing((value) => !value)}
        >
          {editing ? "Done" : "Edit"}
        </Button>
      </div>
      {editing ? (
        <div className="min-h-0 flex-1">
          <SharedTextArea
            keyName={storageKey}
            placeholder="Write notes in markdown"
          />
        </div>
      ) : null}
      {!editing && text ? (
        <div className="flex flex-col gap-2 text-sm leading-relaxed text-foreground">
          <Markdown>{text}</Markdown>
        </div>
      ) : null}
      {!editing && !text ? (
        <p className="text-sm text-muted-foreground">
          No notes yet. Select Edit to write some.
        </p>
      ) : null}
    </div>
  );
}

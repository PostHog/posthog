// Sticky note. The text is a shared field under STORAGE_KEY, so two people can
// type in it at the same time and no character is lost. For a second sticky on
// the same board, change STORAGE_KEY to a different value, such as
// "sticky:risks".
import { SharedTextArea } from "@posthog/canvas-sdk";

const STORAGE_KEY = "sticky:note";

export default function StickyNote({ storageKey = STORAGE_KEY }) {
  return (
    <div className="h-full bg-warning p-3">
      <SharedTextArea keyName={storageKey} placeholder="Write a note" />
    </div>
  );
}

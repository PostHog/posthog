// Shared checklist. The items are a shared list under STORAGE_KEY, so two
// people can add and remove items at the same time and both changes survive.
// For a second checklist on the same board, change STORAGE_KEY to a different
// value, such as "checklist:launch".
import { useSharedList } from "@posthog/canvas-sdk";
import {
  Button,
  Checkbox,
  Heading,
  Input,
  Label,
  Text,
} from "@posthog/quill";
import { useState } from "react";

const STORAGE_KEY = "checklist:items";
const TITLE = "Checklist";

export default function Checklist({ storageKey = STORAGE_KEY, title = TITLE }) {
  const { items, insert, remove, update } = useSharedList(storageKey);
  const [draft, setDraft] = useState("");
  const done = items.filter((item) => item.value && item.value.done).length;

  const addItem = () => {
    const label = draft.trim();
    if (!label) {
      return;
    }
    insert({ label, done: false });
    setDraft("");
  };

  return (
    <div className="flex h-full flex-col gap-3 overflow-auto p-4">
      <div className="flex items-baseline justify-between gap-2">
        <Heading size="lg">{title}</Heading>
        <Text className="text-muted-foreground">
          {done} of {items.length} done
        </Text>
      </div>
      <div className="flex items-center gap-2">
        <Input
          value={draft}
          onChange={(event) => setDraft(event.target.value)}
          onKeyDown={(event) => {
            if (event.key === "Enter") {
              addItem();
            }
          }}
          placeholder="Add an item"
        />
        <Button variant="outline" onClick={addItem}>
          Add
        </Button>
      </div>
      {items.length === 0 ? (
        <Text className="text-muted-foreground">
          No items yet. Add the first one above.
        </Text>
      ) : null}
      <div className="flex flex-col gap-2">
        {items.map((item) => (
          <div key={item.id} className="flex items-center gap-2">
            <Checkbox
              id={item.id}
              checked={Boolean(item.value && item.value.done)}
              onCheckedChange={(checked) =>
                update(item.id, {
                  label: String((item.value && item.value.label) ?? ""),
                  done: Boolean(checked),
                })
              }
            />
            <Label htmlFor={item.id} className="flex-1">
              {String((item.value && item.value.label) ?? "")}
            </Label>
            <Button variant="outline" size="sm" onClick={() => remove(item.id)}>
              Remove
            </Button>
          </div>
        ))}
      </div>
    </div>
  );
}

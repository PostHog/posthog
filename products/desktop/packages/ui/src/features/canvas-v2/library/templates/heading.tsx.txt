// A heading for a group of fragments. Both lines are edited on the board, and
// the text is shared, so everyone sees the same words. Every heading keeps its
// own text, because the keys hold the id of the fragment.
import { useSharedText } from "@posthog/canvas-sdk";

export default function BoardHeading({ fragmentId }) {
  const title = useSharedText("heading:" + fragmentId + ":title");
  const subtitle = useSharedText("heading:" + fragmentId + ":subtitle");

  return (
    <div className="flex h-full flex-col justify-center gap-1">
      <input
        value={title.text}
        placeholder="Board title"
        aria-label="Board title"
        spellCheck={false}
        className="w-full border-0 bg-transparent p-0 font-semibold text-3xl leading-tight tracking-tight outline-none placeholder:text-foreground/25"
        onChange={(event) => title.setText(event.target.value, null)}
      />
      <input
        value={subtitle.text}
        placeholder="Add a line"
        aria-label="Subtitle"
        spellCheck={false}
        className="w-full border-0 bg-transparent p-0 text-muted-foreground text-sm outline-none placeholder:text-foreground/20"
        onChange={(event) => subtitle.setText(event.target.value, null)}
      />
    </div>
  );
}
